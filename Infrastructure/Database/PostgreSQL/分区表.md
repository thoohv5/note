## 分区表

```sql
## ERROR:  cannot ALTER TABLE "comptroller" because it is being used by active queries in this session
CREATE TABLE IF NOT EXISTS comptroller (
    id BIGSERIAL NOT NULL,
    content text NOT NULL DEFAULT '',
    record_time timestamp NOT NULL,
    created_time timestamptz NOT NULL DEFAULT now(),
    updated_time timestamptz NOT NULL DEFAULT now()
) PARTITION BY RANGE ((record_time::date));

## 创建默认子分区
CREATE TABLE comptroller_def PARTITION OF comptroller DEFAULT;

CREATE OR REPLACE FUNCTION func_comptroller_partition() RETURNS trigger LANGUAGE plpgsql AS
$$
DECLARE tbname varchar(64);
DECLARE sql text;
BEGIN
		tbname = 'comptroller_' || to_char(NEW.record_time, 'YYYY_MM');
		sql = format('INSERT INTO %I VALUES ($1.*)', tbname);
		EXECUTE sql USING NEW;
		RETURN NULL;
		EXCEPTION WHEN undefined_table THEN
		BEGIN
        EXECUTE format('CREATE TABLE %I (LIKE comptroller INCLUDING INDEXES)', tbname);
				EXECUTE format('ALTER TABLE comptroller ATTACH PARTITION %I FOR VALUES FROM (%I) TO (%I)', tbname, to_char(NEW.record_time, 'YYYY-MM-01'), to_char(NEW.record_time + interval '1 month', 'YYYY-MM-01'));
		EXCEPTION
        WHEN duplicate_table THEN
						NULL;
    END;
		EXECUTE sql USING NEW;
    RETURN NULL;
END;
$$;

DROP TRIGGER tgr_comptroller_insert ON comptroller;
CREATE TRIGGER tgr_comptroller_insert BEFORE INSERT ON comptroller FOR EACH ROW WHEN (pg_trigger_depth() < 1) EXECUTE FUNCTION func_comptroller_partition();
```
