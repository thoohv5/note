```bash
CREATE TABLE `med_clinic`.`clinic`  (
  `id` int(11) UNSIGNED NOT NULL AUTO_INCREMENT,
  `doctor_id` int(20) UNSIGNED NOT NULL COMMENT '医生userId',
  `patient_id` int(20) UNSIGNED NOT NULL COMMENT '患者userId',
  `group_id` int(11) UNSIGNED NOT NULL DEFAULT 0 COMMENT 'IMId',
  `type` tinyint(4) NOT NULL DEFAULT 0 COMMENT '诊室当前服务类型：0-回合，1-时长次数，默认：0',
  `start_time` int(11) NOT NULL DEFAULT 0 COMMENT '时长次数，服务开始时间',
  `end_time` int(11) NOT NULL DEFAULT 0 COMMENT '时长次数，服务结束时间',
  `created_at` datetime(0) NOT NULL DEFAULT CURRENT_TIMESTAMP(0) COMMENT '创建时间',
  `updated_at` datetime(0) NOT NULL DEFAULT CURRENT_TIMESTAMP(0) ON UPDATE CURRENT_TIMESTAMP(0) COMMENT '更新时间',
  `deleted_at` datetime(0) NULL DEFAULT NULL COMMENT '删除时间',
  PRIMARY KEY (`id`) USING BTREE
) ENGINE = InnoDB AUTO_INCREMENT = 1345 CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci ROW_FORMAT = Compact;
```
