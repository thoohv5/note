## 指令

```bash
## topic list
/opt/kafka/bin/kafka-topics.sh --list --zookeeper zoo1:2181,zoo2:2181,zoo3:2181

/opt/kafka/bin/kafka-topics.sh --list --zookeeper zk:2181

## topic details
/opt/kafka/bin/kafka-topics.sh --describe --zookeeper zoo1:2181,zoo2:2181,zoo3:2181 --topic test

/opt/kafka/bin/kafka-topics.sh --describe --zookeeper zk:2181 --topic test

## create topic
/opt/kafka/bin/kafka-topics.sh --create --topic test --replication-factor 2 --partitions 6 --zookeeper zoo1:2181,zoo2:2181,zoo3:2181 

/opt/kafka/bin/kafka-topics.sh --create --topic consult --replication-factor 1 --partitions 1 --zookeeper zk:2181

## add partitions
/opt/kafka/bin/kafka-topics.sh  --alter --topic gome --partitions 2 --zookeeper zk:2181

## del topic
/opt/kafka/bin/kafka-topics.sh --zookeeper zoo1:2181,zoo2:2181,zoo3:2181 --delete --topic "test"

## create group
/opt/kafka/bin/kafka-consumer-groups.sh --bootstrap-server kafka:9092 --topic consult --group consult-inner

## 发布者
/opt/kafka/bin/kafka-console-producer.sh --broker-list kafka1:9092,kafka2:9092,kafka3:9092 --topic test

/opt/kafka/bin/kafka-console-producer.sh --broker-list kafka:9092 --topic test

## 消费者
/opt/kafka/bin/kafka-console-consumer.sh --bootstrap-server kafka1:9092,kafka2:9092,kafka3:9092 --topic test --from-beginning

/opt/kafka/bin/kafka-console-consumer.sh --bootstrap-server kafka:9092 --topic test --from-beginning

/opt/kafka/bin/kafka-console-consumer.sh --bootstrap-server kafka:9092 --topic consult --from-beg
inning --group consult1

## 消费者群组
/opt/kafka/bin/kafka-consumer-groups.sh --bootstrap-server kafka:9092  --list
/opt/kafka/bin/kafka-consumer-groups.sh --bootstrap-server kafka:9092 --group group_test --describe
/opt/kafka/bin/kafka-consumer-groups.sh --bootstrap-server kafka:9092 --group group_test --describe --members
```
