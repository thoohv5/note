# 自动更新时间戳

```sql
create table users(id serial primary key, created_time timestamptz not null default now(), updated_time timestamptz not null default now());

create or replace function updated_time_column() returns trigger as 
$$ 
begin 
	NEW.updated_time = now();
	return NEW; 
end
$$ 
language 'plpgsql';

create trigger tri_users_update_time before update on public.users for each row execute procedure updated_time_column();
```
