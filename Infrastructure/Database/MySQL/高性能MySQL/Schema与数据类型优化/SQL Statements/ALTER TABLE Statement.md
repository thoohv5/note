# ALTER TABLE Statement

```jsx
CREATE TABLE t1 (c1 INT, c2 INT GENERATED ALWAYS AS (c1 + 1) VIRTUAL);

ALTER TABLE t1 DROP COLUMN c2;
ALTER TABLE t1 ADD COLUMN c2 INT GENERATED ALWAYS AS (c1 + 1) VIRTUAL;
ALTER TABLE t1 MODIFY COLUMN c2 TINYINT GENERATED ALWAYS AS (c1 + 5) VIRTUAL;
ALTER TABLE t1 CHANGE c2 INT GENERATED ALWAYS AS (c1 + 1) VIRTUAL;
```
