# 字段修改

```sql

# 删除字段：
ALTER TABLE table_name DROP COLUMN IF EXISTS field;
# 添加字段：
ALTER TABLE table_name ADD COLUMN field text not null default '';

# 更改字段默认值：
alter table table_name ALTER COLUMN field **SET** DEFAULT '';  
# 添加字段默认值：
ALTER TABLE table_name ADD COLUMN field **INTEGER** DEFAULT 1;
# 删除字段默认值：
ALTER TABLE table_name ALTER COLUMN field **DROP** DEFAULT;

# 删除非空约束
ALTER TABLE table_name ALTER COLUMN field DROP not null;
# 添加非空约束
ALTER TABLE table_name ALTER COLUMN field SET not null;
```
