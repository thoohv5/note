# 基于二进制日志文件位置的复制

1. 配置主数据库服务器【重新启动服务器】
    
    ```bash
    [mysqld]
    server-id=1
    log-bin=master-bin
    log-bin-index=master-bin.index
    
    binlog-do-db #【可选】二进制日志记录的数据库（多数据库用逗号，隔开）
    binlog-ignore-db #【可选】二进制日志中忽略数据库 （多数据库用逗号，隔开）
    
    # 检查
    SHOW VARIABLES LIKE '%server%';
    SHOW VARIABLES LIKE '%log%';
    ```
    
2. 配置从数据库服务【重新启动服务器】
    
    ```bash
    [mysqld]
    server-id=2
    log-bin=master-bin
    log-bin-index=master-bin.index
    relay-log=slave-relay-bin
    relay-log-index=slave-relay-bin.index
    log_save_update=on  #【可选】链路复制
    read_only=on  #【可选】非super用户不能修改
    
    replicate-do-db #【可选】设定需要复制的数据库（多数据库使用逗号，隔开）
    replicate-ignore-db #【可选】设定需要忽略的复制数据库 （多数据库使用逗号，隔开）
    replicate-do-table  #【可选】设定需要复制的表
    replicate-ignore-table #【可选】设定需要忽略的复制表 
    replicate-wild-do-table #【可选】同replication-do-table功能一样，但是可以通配符
    replicate-wild-ignore-table #【可选】同replication-ignore-table功能一样，但是可以加通配符
    
    # 检查
    SHOW VARIABLES LIKE '%server%';
    SHOW VARIABLES LIKE '%log%';
    ```
    
3. 在主数据库服务上建立复制账号
    
    ```bash
    CREATE USER 'repl'@'%.example.com' IDENTIFIED BY 'password';
    GRANT REPLICATION SLAVE ON *.* TO 'repl'@'%.example.com';
    FLUSH PRIVILEGES;
    
    CREATE USER 'repl'@'%' IDENTIFIED BY 'password';
    GRANT REPLICATION SLAVE ON *.* TO 'repl'@'%';
    FLUSH PRIVILEGES;
    
    # 检查
    select Host,User,Repl_slave_priv from mysql.user;
    ```
    
4. 在主数据库上记录需要复制的二进制文件和位置
    
    ```c
    # 刷新所有表并阻止写入语句
    FLUSH TABLES WITH READ LOCK;
    # 释放表
    UNLOCK TABLES;
    
    # 确定当前二进制日志文件的名称和位置
    SHOW MASTER STATUS;
    ```
    
5. 初始化从服务器数据 
    
    ```bash
    
    # 方案一 （InnoDB）
    mysqldump --all-databases --master-data=2 --single-transaction > back_all.sql
    mysql < back_all.sql
    
    # 方案二 【关闭主服务器】
    使用原始数据文件创建数据快照
    
    # 方案三
    xtrabackup --slave info
    
    ```

6. 启动复制链路
    
    ```bash
    CHANGE MASTER TO
    MASTER_HOST='master_host_name',
    MASTER_USER='replication_user_name',
    MASTER_PASSWORD='replication_password',
    MASTER_PORT='master_hostt_port',
    MASTER_LOG_FILE='recorded_log_file_name',
    MASTER_LOG_POS=0;
    
    CHANGE MASTER TO
    MASTER_HOST='172.22.0.3',
    MASTER_USER='repl',
    MASTER_PASSWORD='password',
    MASTER_PORT=3306,
    MASTER_LOG_FILE='master-bin.000002',
    MASTER_LOG_POS=1462;
    ```
    
7. 验证
    
    ```bash
    # 主数据库查看复制状态
    SHOW MASTER STATUS;
    
    # 从数据库查看复制状态
    SHOW SLAVE STATUS;
    
    # 从数据库停止复制
    STOP SLAVE;
    
    # 从数据库开始复制
    START SLAVE;
    ```
    
8. 主从不一致，从数据库直接跳过执行
