# 磁盘扩容

# 命令

```
# 挂载磁盘(硬盘)
mkfs.ext4 /dev/sdb[n]
mount /dev/sdb[n] /data

# 扩展磁盘
fdisk /dev/sda

# 移除分区
d

# 新建分区
n

# 展示分区
p

# 保存变更
w

(reboot system)

# 刷新磁盘
resize2fs /dev/sda3

```
