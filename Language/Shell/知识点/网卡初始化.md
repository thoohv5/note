# 网卡初始化

```bash
#!/bin/bash

# 用法:
# ./set_static_ip_preserve.sh <静态IP> <网关>

IFACE="ens192"
STATIC_IP=$1
GATEWAY=$2

if [ -z "$STATIC_IP" ] || [ -z "$GATEWAY" ]; then
    echo "用法: $0 <静态IP> <网关>"
    echo "例如: $0 192.168.1.100 192.168.1.1"
    exit 1
fi

CONFIG_FILE="/etc/sysconfig/network-scripts/ifcfg-${IFACE}"

# 备份旧配置
cp "$CONFIG_FILE" "${CONFIG_FILE}.bak_$(date +%F_%T)"

# 保留原内容，逐行替换需要的字段
# 生成新配置内容
NEW_CONFIG=""
while IFS= read -r line; do
    case "$line" in
        BOOTPROTO=*)       NEW_CONFIG+="BOOTPROTO=static"$'\n' ;;
        ONBOOT=*)          NEW_CONFIG+="ONBOOT=yes"$'\n' ;;
        IPADDR=*)          NEW_CONFIG+="IPADDR=${STATIC_IP}"$'\n' ;;
        PREFIX=*)          NEW_CONFIG+="PREFIX=24"$'\n' ;;
        GATEWAY=*)         NEW_CONFIG+="GATEWAY=${GATEWAY}"$'\n' ;;
        DNS1=*|DNS2=*)     continue ;;  # 删除原有 DNS，后面再加新的
        IPADDR=*|PREFIX=*|GATEWAY=*) continue ;;  # 清除重复字段
        *)                 NEW_CONFIG+="$line"$'\n' ;;
    esac
done < "$CONFIG_FILE"

# 添加或替换字段
[[ "$NEW_CONFIG" != *"IPADDR="* ]] && NEW_CONFIG+="IPADDR=${STATIC_IP}"$'\n'
[[ "$NEW_CONFIG" != *"PREFIX="* ]] && NEW_CONFIG+="PREFIX=24"$'\n'
[[ "$NEW_CONFIG" != *"GATEWAY="* ]] && NEW_CONFIG+="GATEWAY=${GATEWAY}"$'\n'
NEW_CONFIG+="DNS1=114.114.114.114"$'\n'
NEW_CONFIG+="DNS2=8.8.8.8"$'\n'

# 写回配置文件
echo "$NEW_CONFIG" > "$CONFIG_FILE"

# 重启网卡
echo "正在重启网卡 ${IFACE}..."
ifdown ${IFACE} 2>/dev/null
ifup ${IFACE} 2>/dev/null

# NetworkManager reload（如存在）
if systemctl is-active --quiet NetworkManager; then
    nmcli connection reload
    nmcli connection down "$IFACE" || true
    nmcli connection up "$IFACE" || true
fi

echo "✅ 网卡配置完成！"
echo "接口：$IFACE"
echo "IP地址：$STATIC_IP"
echo "网关：$GATEWAY"
echo "DNS：114.114.114.114, 8.8.8.8"

```
