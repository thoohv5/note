# 接受编译参数

```

# 构建golang实现的适配插件 # example: ./build_plugin.sh -p ./h3c/S5120V2-28P-LI/main.go -n h3c_S5120V2

# Step 1: 接收编译参数
while getopts "p:n:" opt; do
  case $opt in
    p) path=$OPTARG ;;
    n) name=$OPTARG ;;
    *) echo "Invalid option"; exit 1 ;;
  esac
done
```
