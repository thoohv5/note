## 组合（虚拟继承）

Go 的方法调用没有 Java 那样的虚方法（virtual function）机制，不会运行时再去查找是否有 “覆盖的方法”。

```go
package main

import "fmt"

type IBase[S, T any] interface {
	ToEntity(s S) T
	ToEntities(ss []S) []T
}
type Base[S, T any] struct {
	IBase[S, T]
}

func (b *Base[S, T]) ToEntity(s S) T {
	fmt.Println("Base Entity")
	var t T
	return t
}

func (b *Base[S, T]) ToEntities(ss []S) []T {
	for _, s := range ss {
		if b.IBase == nil {
			b.ToEntity(s)
			continue
		}
		b.IBase.ToEntity(s)
	}
	fmt.Println("Base Entities")
	return nil
}

type Current[S, T any] struct {
	Base[S, T]
}

func (c *Current[S, T]) ToEntity(s S) T {
	fmt.Println("Current Entity")
	var t T
	return t
}
```
