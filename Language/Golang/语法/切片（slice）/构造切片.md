# 构造切片

```

s := make([]int32, 0, 200)
for i := 0; i < 200; i++ {
    s = append(s, int32(i+1))
}
ptr := unsafe.Pointer(&s[0])
fmt.Println(s)
fmt.Println(ptr)

// 构造slice： 方式一
s1 := ((*[1 << 10]int32)(ptr))[:200]
fmt.Printf("方式一:%v\\n", s1)

// 构造slice： 方式二
s2 := struct {
    addr uintptr
    len  int
    cap  int
}{uintptr(ptr), 200, 200}
s21 := *(*[]int32)(unsafe.Pointer(&s2))
fmt.Printf("方式二:%v, %v\\n", s2, s21)

// 构造slice： 方式三
var s3 []int32
sliceHeader := (*reflect.SliceHeader)(unsafe.Pointer(&s3))
sliceHeader.Cap = 200
sliceHeader.Len = 200
sliceHeader.Data = uintptr(ptr)
fmt.Printf("方式三:%v\\n", s3)

// 构造slice： 方式四
usd := unsafe.SliceData(unsafe.Slice((*int32)(ptr), 200))
s4 := *(*[]int32)(unsafe.Pointer(&usd))
fmt.Printf("方式四:%v\\n", s4[:200])

```
