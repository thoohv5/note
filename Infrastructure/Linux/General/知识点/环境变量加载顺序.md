# 环境变量加载顺序

```bash
# login shell
/etc/profile => /etc/profile.d/*.sh => ~/.bash_profile => ~/.bashrc => /etc/bashrc
# no login shell
/etc/bashrc => /etc/profile.d/*.sh

```

![](https://raw.githubusercontent.com/thoohv5/ob/main/pictureenv_file.png)

```bash
# login shell环境变量（全局）
/etc/profile
# 作用
1. USER变量
2. LOGNAME变量
3. MAIL变量
4. PATH变量
5. HOSTNAME变量
6. HISTIZE变量
7. umask

```

```bash
# no login shell环境变量（全局）
/etc/bashrc
# 自定义脚本（全局）
/etc/profile.d/xxx.sh
# 作用
1. PS1变量

```

```bash

# 环境变量（当前用户）
~/.bashrc_profile
# 作用
1. PATH变量

```

```bash
# 环境变量（当前用户）
~/.bashrc
# 作用
1. 别名

```

```bash
# 显示当前shell的所有变量（环境变量、临时变量）
set/declare
# 当前用户的环境变量
env
# 用来显示和设置导出到子shell的环境变量
export
# 删除
unset
# 只读环境变量
readonly

```

# 特殊配置文件

```bash
# 登录退出文件
~/.bash_logout

```

```bash
# 历史命令记录文件
~/bash_history

```

```bash
# 本地终端欢迎信息
/etc/issue

# 远程终端环境信息
/etc/issue.net

# 本地、远程
/etc/motd
```
