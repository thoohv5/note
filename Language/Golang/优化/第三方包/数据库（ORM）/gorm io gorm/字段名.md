## 字段名

```go
import (
    "reflect"
    "strings"
    "unicode"
    "gorm.io/gorm"
)

// 获取结构体的数据库列名（自动处理嵌套结构体和 gorm 标签）
func GetModelColumnNames(model interface{}) ([]string, error) {
    var columns []string
    val := reflect.ValueOf(model)
    
    // 如果是指针，获取其指向的值
    if val.Kind() == reflect.Ptr {
        val = val.Elem()
    }
    
    // 必须是一个结构体
    if val.Kind() != reflect.Struct {
        return nil, fmt.Errorf("input must be a struct or struct pointer")
    }
    
    // 递归遍历结构体字段
    processStruct(val.Type(), &columns)
    return columns, nil
}

// 递归处理结构体及其嵌套字段
func processStruct(t reflect.Type, columns *[]string) {
    for i := 0; i < t.NumField(); i++ {
        field := t.Field(i)
        
        // 跳过非导出字段（小写开头的字段）
        if !field.IsExported() {
            continue
        }
        
        // 处理嵌套结构体（如 gorm.Model）
        if field.Anonymous && field.Type.Kind() == reflect.Struct {
            processStruct(field.Type, columns)
            continue
        }
        
        // 解析 gorm 标签中的列名
        tag := field.Tag.Get("gorm")
        if tag == "-" {
            continue // 跳过被忽略的字段
        }
        
        columnName := parseGormTag(tag)
        if columnName == "" {
            // 无自定义列名，生成默认的 snake_case
            columnName = toSnakeCase(field.Name)
        }
        
        *columns = append(*columns, columnName)
    }
}

// 解析 gorm 标签中的 column 名称
func parseGormTag(tag string) string {
    for _, part := range strings.Split(tag, ";") {
        if strings.HasPrefix(part, "column:") {
            return strings.TrimPrefix(part, "column:")
        }
    }
    return ""
}

// 将驼峰字段名转为 snake_case（如 UserID → user_id）
func toSnakeCase(s string) string {
    var snake []rune
    for i, r := range s {
        if i > 0 && unicode.IsUpper(r) {
            snake = append(snake, '_')
        }
        snake = append(snake, unicode.ToLower(r))
    }
    return string(snake)
}
```
